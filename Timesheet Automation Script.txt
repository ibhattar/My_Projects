// ── Trigger entry points ──────────────────────────────────────────────────────

/** Runs every Friday via a weekly time-based trigger */
function weeklyFridayCheck() {
  checkAllTimesheets();
}

/** Runs daily; only proceeds on the last day of the month */
function endOfMonthCheck() {
  if (isLastDayOfMonth()) {
    Logger.log("End of month detected – running timesheet check.");
    checkAllTimesheets();
  }
}

// ── Helper ────────────────────────────────────────────────────────────────────

function isLastDayOfMonth() {
  const today    = new Date();
  const tomorrow = new Date(today);
  tomorrow.setDate(today.getDate() + 1);
  // If tomorrow is the 1st, today is the last day of the month
  return tomorrow.getDate() === 1;
}

// ── One-time setup: call this ONCE from the Apps Script editor ────────────────

/**
 * Creates two triggers:
 *   1. weeklyFridayCheck  – every Friday
 *   2. endOfMonthCheck    – daily (checks internally if it's the last day)
 *
 * Run setupTriggers() once manually. Delete any old triggers first to avoid duplicates.
 */
function setupTriggers() {
  // Remove all existing triggers to avoid duplicates
  ScriptApp.getProjectTriggers().forEach(t => ScriptApp.deleteTrigger(t));

  // 1. Weekly – every Friday
  ScriptApp.newTrigger("weeklyFridayCheck")
    .timeBased()
    .onWeekDay(ScriptApp.WeekDay.FRIDAY)
    .atHour(9)          // 9 AM in the script's timezone
    .create();

  // 2. Daily – for end-of-month detection
  ScriptApp.newTrigger("endOfMonthCheck")
    .timeBased()
    .everyDays(1)
    .atHour(9)          // same time as Friday run
    .create();

  Logger.log("Triggers created successfully.");
}
